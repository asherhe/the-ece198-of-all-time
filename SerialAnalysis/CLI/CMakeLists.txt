cmake_minimum_required(VERSION 3.20)
project(serial_analysis LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

add_executable(serial-analysis main.cpp)
add_executable(clock-probe clock_probe.cpp)
foreach(target serial-analysis clock-probe)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra)
  endif()
  if(WIN32)
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  endif()
endforeach()

enable_testing()
add_test(NAME self-test COMMAND serial-analysis --self-test)
